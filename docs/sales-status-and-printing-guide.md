# แนวทางระบบสถานะใบขายและการพิมพ์เอกสาร

สรุปจากระบบที่ใช้งานจริงใน Hippo Village เพื่อให้ **Ranger TOPSHOP** นำไปทำตาม

เอกสารนี้เขียนเป็น "หลักการ + เหตุผลว่าทำไม" ไม่ใช่โค้ดให้ก๊อป โค้ดตัวอย่างเป็น Google Apps Script
แต่หลักการใช้ได้กับทุกแพลตฟอร์ม ทุกข้อในนี้มีที่มาจากของที่เคยพังจริงแล้วทั้งนั้น

---

# ส่วนที่ 1 — ระบบสถานะใบขาย

## 1.1 ลำดับสถานะต้องเป็นข้อมูล ไม่ใช่ `if` กระจาย

ประกาศลำดับไว้ที่เดียวเป็นอาเรย์ แล้วให้ทุกอย่างอ้างจากตรงนั้น

```js
var SO_FLOW = ['open', 'confirmed', 'picking', 'ready_to_ship', 'shipped'];
var SO_STATUS_LABEL = {
  open: 'เปิดการขาย', confirmed: 'ยืนยันแล้ว', picking: 'เริ่มหยิบ',
  ready_to_ship: 'รอส่ง', shipped: 'ส่งแล้ว', cancelled: 'ยกเลิก'
};
```

การตรวจว่าเดินสถานะได้ไหม จึงเหลือสองบรรทัด

```js
var ci = SO_FLOW.indexOf(cur), ti = SO_FLOW.indexOf(to);
if (ti === -1) return { success: false, message: 'สถานะไม่ถูกต้อง' };
if (ti !== ci + 1) return { success: false, message: 'ต้องเดินสถานะทีละขั้นตามลำดับ' };
```

**ทำไม:** ถ้าเขียนเป็น `if (cur === 'open' && to === 'confirmed') ... else if ...` วันที่ต้องแทรกสถานะใหม่
ตรงกลาง จะต้องไล่แก้ทุกที่ที่เคยเขียนเงื่อนไขไว้ แล้วจะมีที่ใดที่หนึ่งถูกลืมเสมอ

**ห้ามข้ามขั้น** แม้แอดมินก็ห้าม — ถ้าข้ามได้ ขั้นที่มีผลข้างเคียง (จองของ/ตัดของ) จะถูกข้ามไปด้วย
แล้วตัวเลขสต็อกจะเพี้ยนโดยไม่มีอะไรฟ้อง

## 1.2 ★★★ จุดตัดสต็อกจริงอยู่ตรงไหน คือการตัดสินใจที่สำคัญที่สุดของทั้งระบบ

ใน Hippo Village ตัดที่ **"รอส่ง" (ready_to_ship)** ไม่ใช่ตอน "ส่งแล้ว"

```
open          เปิดใบ         ไม่แตะสต็อกเลย
confirmed     ยืนยัน         → จองของ (reserve)  ของยังอยู่ในคลัง
picking       เริ่มหยิบ       ไม่แตะสต็อก แค่บอกว่าใครกำลังทำอยู่
ready_to_ship รอส่ง          → ★ ตัด on-hand จริง + ปลดจองพร้อมกัน
shipped       ส่งแล้ว         บันทึกเวลาอย่างเดียว ห้ามตัดซ้ำ
```

**เหตุผล:** ของถูกหยิบออกจากชั้นและแพ็คใส่กล่องแล้ว มันไม่ได้อยู่ในคลังอีกต่อไปตั้งแต่ตอนนั้น
ถ้ารอตัดตอน "ส่งแล้ว" ระบบจะบอกว่ามีของอยู่ ทั้งที่มันนอนอยู่ในกล่องหน้าร้าน — แล้วจะมีคนมาขายซ้ำ

**สำหรับ TOPSHOP:** งานขายรถเร่จุดตัดอาจต่างจากนี้ (เช่นตัดตอนโหลดของขึ้นรถ) **แต่ต้องเลือกให้ชัด
แล้วเขียนกำกับไว้** และต้องมีจุดตัดเพียงจุดเดียวในทั้งระบบ

**ห้ามตัดซ้ำในขั้นถัดไปเด็ดขาด** — บั๊กนี้ทำให้ใบขายใบเดียวหักสต็อกสองเท่า และมองไม่เห็นจากหน้าจอเลย

## 1.3 สต็อกต้องมีสามตัวเลข ไม่ใช่ตัวเดียว

```
on-hand    = ของที่อยู่ในคลังจริง                    (branch_stock.qty)
reserved   = ของที่ถูกจองให้ใบขายอื่นที่ยังไม่ตัด      (stock_reservations status='active')
available  = on-hand − reserved                    ← ตัวที่ใช้ตัดสินว่าขายได้ไหม
```

```js
function getAvailableQty(branchId, warehouseId, itemType, itemId) {
  return getBranchStockQty(branchId, warehouseId, itemType, itemId)
       - getReservedQty(branchId, warehouseId, itemType, itemId);
}
```

**ถ้าไม่มี reserved** สองใบขายจะยืนยันของชิ้นเดียวกันได้พร้อมกัน แล้วไปเจอตอนแพ็คว่าของไม่พอ
ซึ่งตอนนั้นสัญญากับลูกค้าไปแล้วทั้งสองราย

**การจองเก็บเป็นแถวในตารางแยก** ไม่ใช่คอลัมน์ในตารางสต็อก — เพราะต้องรู้ว่าใครจอง จองให้ใบไหน
และปลดจองทีละใบได้ ถ้าเก็บเป็นตัวเลขรวม พอมีอะไรผิดจะแกะไม่ออกว่าผิดที่ใบไหน

**ปลดจองพร้อมกับตอนตัดของเสมอ** ไม่งั้นของจะถูกนับสองทาง (หายจาก on-hand แล้วยังกันยอดจองไว้อีก)

## 1.4 การยกเลิกมีสองแบบคนละเรื่องกัน

```js
var SO_STOCK_TAKEN = ['ready_to_ship', 'shipped'];
function _soCancelReturnsStock(status) { return SO_STOCK_TAKEN.indexOf(String(status)) >= 0; }
```

- **ยังไม่ถึงจุดตัด** → ของยังอยู่ในคลัง แค่ปลดจองก็พอ
- **เลยจุดตัดไปแล้ว** → ของถูกตัดไปแล้ว **ต้องคืนกลับเข้าคลัง**

**★ ค่าคงที่ตัวนี้สำคัญกว่าที่คิด** — มันทำให้ "จุดที่ตัดของ" กับ "จุดที่คืนของ" อ้างนิยามเดียวกัน
วันที่ย้ายจุดตัดสต็อก แก้ที่อาเรย์ตัวนี้ตัวเดียว ทั้งสองฝั่งขยับตามพร้อมกัน

ถ้าไม่คืนของ: ของจะหายจากระบบถาวรทั้งที่ยังวางอยู่หน้าร้าน และ**ไม่มีอะไรเตือนเลย**
เพราะยอดขายกับยอดสต็อกลดพร้อมกันดูสมเหตุสมผลทุกอย่าง

## 1.5 สถานะปลายทางต้องล็อก

```js
if (cur === 'shipped')   return { success:false, message:'ใบขายนี้จัดส่งแล้ว แก้สถานะไม่ได้' };
if (cur === 'cancelled') return { success:false, message:'ใบขายนี้ถูกยกเลิกแล้ว' };
```

และ**แก้ไขรายการสินค้าได้เฉพาะสถานะแรกสุดเท่านั้น** — พอยืนยันแล้วมีการจองของเกิดขึ้น
การแก้รายการทีหลังแปลว่าต้องไปแก้ยอดจองตามด้วย ซึ่งเป็นจุดที่พลาดง่ายที่สุด
ปิดประตูไปเลยง่ายกว่าและปลอดภัยกว่า

## 1.6 บันทึกประวัติต้องเก็บ "ชื่อ ณ เวลานั้น" ไม่ใช่รหัสอ้างอิง

งานใบขายเดินข้ามแผนก — ฝ่ายขายเปิดใบ คลังหยิบของ ฝ่ายจัดส่งส่งของ จึงต้องรู้ว่าใครลงมือขั้นไหน

**เก็บชื่อและตำแหน่งลงไปในแถว log เลย** ไม่ใช่เก็บแค่ `user_id` แล้วไปอ่านชื่อตอนแสดงผล
เพราะคนเปลี่ยนชื่อได้ ย้ายแผนกได้ ลาออกได้ — **หลักฐานต้องคงสภาพเดิม**

บันทึกทุกครั้งที่: สร้างใบ · แก้รายการ · เปลี่ยนสถานะ · ทำเอกสารเกี่ยวเนื่อง

## 1.7 ชื่อสถานะต้องมีแหล่งความจริงแหล่งเดียว

หน้าบ้านกับหลังบ้านต้องใช้ชื่อชุดเดียวกันเป๊ะ ถ้าหลังบ้านเก็บ `ready_to_ship` แต่หน้าบ้านเขียน
`readyToShip` จะกลายเป็นคนละสถานะทันทีและหาไม่เจอว่าทำไมปุ่มไม่โผล่

**แยก "ป้ายข้อความ" ออกจาก "สีที่แสดง"** เป็นคนละ map กัน

```js
const SO_STATUS_LABEL = { open:'เปิดการขาย', confirmed:'ยืนยันแล้ว', ... };
const SO_TAG          = { open:'draft', confirmed:'ordered', picking:'ordered',
                          ready_to_ship:'in_transit', shipped:'received', cancelled:'cancelled' };
```

หลายสถานะใช้สีเดียวกันได้ (`confirmed` กับ `picking` ต่างเป็น `ordered`) การรวมสองเรื่องนี้
ไว้ใน map เดียวจะบังคับให้ทุกสถานะต้องมีสีของตัวเองโดยไม่จำเป็น

## 1.8 ปุ่ม "ขั้นถัดไป" สร้างจากข้อมูล ไม่ใช่เขียนปุ่มทีละอัน

```js
const SO_NEXT = {
  open:          { to:'confirmed',     label:'ยืนยัน' },
  confirmed:     { to:'picking',       label:'เริ่มหยิบ' },
  picking:       { to:'ready_to_ship', label:'แพ็กเสร็จ → ตัดสต็อก' },
  ready_to_ship: { to:'shipped',       label:'ส่งของ' }
};
```

**★ ข้อความบนปุ่มต้องบอกผลที่จะเกิดขึ้นจริง** — ปุ่มที่จะไปตัดสต็อกเขียนว่า
*"แพ็กเสร็จ → ตัดสต็อก"* ไม่ใช่ *"ขั้นถัดไป"* คนกดต้องรู้ตัวก่อนว่ากดแล้วของจะหายจากคลัง

## 1.9 เซิร์ฟเวอร์ตัดสินทุกอย่าง หน้าจอแค่แสดงผล

หน้าจอซ่อนปุ่มได้ แต่**ห้ามเชื่อว่าการซ่อนปุ่มคือการป้องกัน** ทุกกฎในข้อ 1.1–1.5
ต้องบังคับใหม่ทั้งหมดที่ฝั่งเซิร์ฟเวอร์ เพราะใครก็ยิงคำขอตรงเข้ามาได้

# ส่วนที่ 2 — ระบบการพิมพ์เอกสาร

## 2.1 เครื่องพิมพ์ตัวเดียว เอกสารทุกชนิดใช้ร่วมกัน

Hippo Village มีเครื่องพิมพ์สองตัวเท่านั้นสำหรับเอกสารทั้งระบบ

| ตัว | ใช้กับ |
|---|---|
| **ตัวเอกสาร** (`showDocPreview`) | ใบสั่งขาย · ใบส่งสินค้า · ใบเสร็จ · ใบรับของ · ใบจ่ายออก · ใบโอน |
| **ตัวรายงานตาราง** (`_printTableReport`) | รายงานสรุปทุกชนิดที่เป็นตารางยาว ๆ |

**ห้ามเขียนโค้ดพิมพ์แยกต่อเอกสารหนึ่งชนิด** — เอกสารใหม่ = ส่งข้อมูลรูปแบบเดียวกันเข้าเครื่องเดิม
ไม่งั้นพอต้องแก้หัวกระดาษหรือขนาดฟอนต์ จะต้องไล่แก้ทุกไฟล์แล้วลืมบางอัน

## 2.2 ★★★ แบ่งหน้าด้วยการวัดความสูงจริง ห้ามเดาจำนวนแถว

นี่คือข้อที่สำคัญที่สุดของส่วนการพิมพ์

**วิธีที่ผิด:** `const ROWS_PER_PAGE = 20;` แล้วหั่นทุก 20 แถว
→ ชื่อสินค้ายาวขึ้นบรรทัดที่สอง แถวสูงขึ้น หน้าล้น **แถวสุดท้ายหายไปเงียบ ๆ**
ไม่มี error ไม่มีอะไรเตือน เอกสารพิมพ์ออกมาสวยงามแต่ข้อมูลไม่ครบ

**วิธีที่ถูก:** สร้างกล่องวัดที่มีขนาด A4 เท่ากับของจริงเป๊ะ แล้วไล่ใส่แถวทีละแถว วัดทุกครั้ง ล้นเมื่อไหร่ถอยกลับหนึ่งแถวแล้วขึ้นหน้าใหม่

```js
m.className = 'docsheet doc-measure';       // ★ ใช้คลาสเดียวกับของจริง
const overflow = () => m.scrollHeight > m.clientHeight + 1;

while (idx < lines.length) {
  const tr = document.createElement('tbody');
  tr.innerHTML = _docRowHtml(lines[idx], ...);
  tbody.appendChild(tr.firstChild);
  if (overflow()) { tbody.removeChild(tbody.lastChild); break; }   // ถอยกลับ
  take.push(idx++);
}
```

**สามข้อที่ต้องระวัง:**

1. **กล่องวัดต้องซ่อนด้วย `position`/`visibility` ไม่ใช่ `display:none`** — `display:none` ทำให้
   ความสูงทุกอย่างเป็น 0 วัดอะไรไม่ได้เลย
2. **ต้องมีตัวกันวนไม่จบ** — ถ้าแถวเดียวก็ยังล้น (ชื่อยาวผิดปกติ) ต้องยัดลงหน้านั้นไปเลย
   ไม่งั้นลูปไม่มีวันจบและเบราว์เซอร์ค้าง
3. **หน้าสุดท้ายต้องเผื่อที่ให้ยอดรวมกับกลุ่มลงชื่อ** ซึ่งสูงกว่าบรรทัดยอดยกไปของหน้ากลาง
   ถ้าไม่พอต้องผลักแถวท้ายไปหน้าใหม่ทีละแถวจนพอ

## 2.3 หน้ากลางต้องมียอดยกไป/ยกมา

เอกสารหลายหน้าที่ไม่มียอดยกไป คนอ่านจะรวมเลขเองไม่ได้และไม่รู้ว่าได้กระดาษครบหรือยัง
ท้ายทุกหน้า (ยกเว้นหน้าสุดท้าย) ใส่ยอดสะสม และหัวหน้าถัดไปใส่ยอดยกมา
พร้อมเลข **"หน้า X / Y"** ทุกแผ่น

## 2.4 ช่องลงชื่อเป็นข้อมูล ไม่ใช่ HTML ที่เขียนซ้ำในแต่ละแบบ

```js
const DOC_SIGNERS = {
  salesorder: ['ผู้สั่งซื้อ', 'ผู้ขาย', 'ผู้อนุมัติ'],
  delivery:   ['ผู้จัดของ', 'ผู้จัดส่ง', 'ผู้รับสินค้า'],
  invoice:    ['ผู้ขาย', 'ผู้ชำระเงิน'],
  receipt:    ['ผู้ส่งของ', 'ผู้ตรวจรับ', 'ผู้รับเข้าคลัง', 'ผู้อนุมัติ'],
  issue:      ['ผู้ขอเบิก', 'ผู้อนุมัติ', 'ผู้จ่ายของ', 'ผู้รับของ']
};
```

**ช่องลงชื่อสะท้อนหน้าที่ทางธุรกิจของเอกสารนั้น** ไม่ใช่ของประดับ — ใบจ่ายออกต้องมีทั้งคนขอ
และคนอนุมัติ เพราะไม่มีใบสั่งซื้อหรือใบขายมาเป็นต้นเรื่องให้ตรวจย้อน ต่างจากใบส่งสินค้า
ที่มีใบขายเป็นต้นเรื่องอยู่แล้ว

## 2.5 ใบเดียวกันออกได้หลายเอกสารที่มีหน้าที่ต่างกัน

จากใบขายใบเดียว Hippo Village ออกได้สามอย่าง

| เอกสาร | มีราคาไหม | ใครถือ |
|---|---|---|
| ใบสั่งขาย | ✅ มี | ลูกค้า (ยืนยันการสั่งซื้อ) |
| **ใบส่งสินค้า** | ❌ **ไม่มี** | คนขับ + ผู้รับปลายทาง |
| ใบเสร็จรับเงิน | ✅ มี | ลูกค้า (หลักฐานชำระเงิน) |

**★ ใบส่งสินค้าไม่พิมพ์ราคาโดยตั้งใจ** — เอกสารใบนี้ผ่านมือคนขับและผู้รับปลายทาง
ซึ่งไม่ใช่คู่สัญญาการค้า ราคาที่ตกลงกันไม่ควรหลุดไปกับของ

**สำหรับ TOPSHOP ที่เป็นงานรถเร่ เรื่องนี้สำคัญมาก** — ของผ่านมือคนนอกเยอะกว่าขายหน้าร้านมาก


กดปุ่ม → เรียก API → รอ 2-3 วินาที → คอลแบ็กกลับมา → `window.open()` → **เบราว์เซอร์บล็อก**
เพราะ user activation หมดอายุไปแล้ว ผู้ใช้เห็นแค่ "ไม่มีอะไรเกิดขึ้น"

**วิธีที่ใช้ได้จริง:** สร้าง Blob แล้วเอา URL ไปใส่ `href` ของลิงก์ที่มีอยู่แล้วบนหน้า
ให้ผู้ใช้กดลิงก์นั้นเอง — การกดของผู้ใช้พา activation มาด้วยเสมอ

```js
if (_DOC_BLOB) URL.revokeObjectURL(_DOC_BLOB);        // คืนหน่วยความจำของใบก่อน
_DOC_BLOB = URL.createObjectURL(new Blob([html], {type:'text/html'}));
document.getElementById('docPrintTab').href = _DOC_BLOB;
```

## 2.7 พรีวิวกับไฟล์ที่เปิดแท็บใหม่ต้องหน้าตาเหมือนกันเป๊ะ

ไฟล์ที่เปิดแท็บใหม่เป็นเอกสารคนละหน้า **ไม่ได้โหลด CSS ของแอป** ถ้าไม่จัดการ พรีวิวจะสวย
แต่ของที่พิมพ์จริงออกมาเป็นข้อความเปล่า ๆ

วิธีที่ใช้: ไล่เก็บกฎ CSS จาก stylesheet ของแอปเองตอนรันไทม์ แล้วฝังลงไฟล์

```js
for (let i=0; i<document.styleSheets.length; i++){
  let rules; try { rules = document.styleSheets[i].cssRules; } catch(e){ continue; }
  for (let j=0; j<rules.length; j++){
    const t = rules[j].cssText || '';
    if (t.indexOf('.docsheet') >= 0 || t.indexOf(':root') === 0) sheetCss += t + '\n';
  }
}
```

**อย่าลืมประกาศ CSS variable ซ้ำในไฟล์ด้วย** — ไฟล์แยกไม่ได้ `:root` ของแอป ตัวแปรจะตกไปใช้
ค่า fallback แล้วขนาดฟอนต์เพี้ยนทั้งใบ

`try/catch` รอบ `cssRules` จำเป็น เพราะ stylesheet ข้ามโดเมน (เช่น Google Fonts) อ่านไม่ได้และจะโยน error

## 2.8 กฎ CSS สำหรับพิมพ์

```css
@page { size: A4; margin: 14mm }
.docsheet { height: calc(297mm - 28mm); overflow: hidden }
.docsheet + .docsheet { page-break-before: always; break-before: page }
```

**★ ใช้ `.docsheet + .docsheet` ไม่ใช่ `.docsheet`** — ถ้าสั่งตัดหน้าท้ายทุกแผ่นรวมแผ่นสุดท้าย
จะได้**กระดาษเปล่าเพิ่มมาอีกหนึ่งใบทุกครั้งที่พิมพ์**

**★★ ระวัง media query ที่ใช้หน่วย mm** — ตอนพิมพ์ viewport กว้างประมาณ 816px
`@media (max-width: 230mm)` จะทำงานตอนพิมพ์ด้วย แล้ว layout ที่ออกแบบไว้สำหรับมือถือ
จะไปโผล่บนกระดาษ (เคยทำให้ตารางสามคอลัมน์ยุบเหลือคอลัมน์เดียวบนกระดาษจริง)
ถ้าตั้งใจให้ใช้เฉพาะบนจอ ต้องครอบด้วย `@media screen` ให้ชัด

## 2.9 ส่งออกเป็นไฟล์ตาราง ใช้ .xlsx ไม่ใช่ CSV

CSV กับภาษาไทยเปิดใน Excel แล้วเป็นตัวยึกยือทุกครั้ง และรหัสสินค้าที่ขึ้นต้นด้วยศูนย์
จะถูกกลืนเลขศูนย์หน้าทิ้ง

ใช้ไลบรารีเขียน `.xlsx` แล้ว**บังคับชนิดเซลล์เป็นข้อความ** สำหรับคอลัมน์ที่เป็น
"ตัวเลขแต่ไม่ใช่จำนวน" — รหัสสินค้า เลขผู้เสียภาษี เบอร์โทร รหัสไปรษณีย์

## สรุปสามข้อที่ถ้าจำได้แค่นี้ก็พอ
